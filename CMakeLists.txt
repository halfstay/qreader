cmake_minimum_required(VERSION 3.10)
project(QReader VERSION 1.2.0 LANGUAGES C CXX)

option(QREADER_BUILD_TESTS "Build functional tests" ON)
option(QREADER_BUILD_GUI "Build the native Windows application" ON)
if(QREADER_BUILD_GUI AND NOT MINGW)
    message(FATAL_ERROR "Use MinGW-w64 for the GUI, or -DQREADER_BUILD_GUI=OFF for native core tests.")
endif()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(MINGW)
    add_definitions(-D_WIN32_WINNT=0x0601 -DWINVER=0x0601 -DNOMINMAX)
    # Strip as each Release binary is linked, including dependency DLLs and
    # test executables. Set before add_subdirectory; compatible with CMake 3.10
    # and both single- and multi-configuration generators.
    foreach(kind EXE SHARED MODULE)
        set(CMAKE_${kind}_LINKER_FLAGS_RELEASE "${CMAKE_${kind}_LINKER_FLAGS_RELEASE} -Wl,--strip-all")
    endforeach()
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/Dependencies.cmake")
qreader_dependencies()

function(qreader_target target)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    # CMake 3.10 does not know CXX_STANDARD=20. Pass the GCC flag explicitly.
    target_compile_options(${target} PRIVATE -std=c++20 -Wall -Wextra -Wpedantic)
    target_compile_definitions(${target} PRIVATE ZX_USE_UTF8)
    if(MINGW)
        target_compile_definitions(${target} PRIVATE UNICODE _UNICODE)
        set_property(TARGET ${target} APPEND_STRING PROPERTY LINK_FLAGS " -static -static-libgcc -static-libstdc++")
    endif()
endfunction()

add_library(qreader_core STATIC src/analyzer.cpp src/geometry.cpp)
qreader_target(qreader_core)
target_include_directories(qreader_core PUBLIC src)
target_include_directories(qreader_core SYSTEM PUBLIC ${QREADER_OPENCV_INCLUDES})
target_link_libraries(qreader_core PUBLIC opencv_objdetect opencv_imgcodecs opencv_imgproc ZXing::ZXing)

if(QREADER_BUILD_GUI)
    enable_language(RC)
    include("${CMAKE_CURRENT_SOURCE_DIR}/Localization.cmake")
    add_library(qreader_i18n STATIC src/i18n.cpp)
    qreader_target(qreader_i18n)
    target_include_directories(qreader_i18n PUBLIC src)
    target_link_libraries(qreader_i18n PUBLIC advapi32)
    add_library(qreader_windows STATIC src/clipboard.cpp src/window_hit.cpp)
    qreader_target(qreader_windows)
    target_link_libraries(qreader_windows PUBLIC qreader_core qreader_i18n user32 gdi32 shell32 dwmapi)
    add_executable(qreader WIN32 src/main.cpp src/capture.cpp src/win_util.cpp res/qreader.rc "${QREADER_LOCALE_RC}")
    qreader_target(qreader)
    target_include_directories(qreader PRIVATE res)
    target_link_libraries(qreader PRIVATE qreader_windows comctl32 comdlg32)
    set_property(TARGET qreader APPEND_STRING PROPERTY LINK_FLAGS " -municode -Wl,--major-subsystem-version,6,--minor-subsystem-version,1")
    install(TARGETS qreader RUNTIME DESTINATION .)
    install(FILES LICENSE README.md THIRD_PARTY_NOTICES.md TESTING.md CONTRIBUTING.md DESTINATION .)
    file(GLOB QREADER_TRANSLATED_READMES "${CMAKE_CURRENT_SOURCE_DIR}/docs/README.*.md")
    install(FILES ${QREADER_TRANSLATED_READMES} docs/I18N.md DESTINATION docs)
    install(FILES res/i18n/territories.csv res/i18n/UNICODE-LICENSE.txt DESTINATION res/i18n)
    install(FILES res/i18n/UNICODE-LICENSE.txt DESTINATION licenses)
    install(DIRECTORY "${CMAKE_BINARY_DIR}/licenses/" DESTINATION licenses)
endif()
